4.1 Gráfica combinada: rendimiento anual del activo vs. crecimiento del PIB

# Para el rendimiento anual

rend_anual <- apply.yearly(
  simple_rend,
  function(x) mean(x) * 252 
)

rend_anual <- data.frame(
  anio = as.numeric(format(index(rend_anual), "%Y")),
  rendimiento = as.numeric(rend_anual) * 100
)

head(rend_anual)
##   anio rendimiento
## 1 2000   -58.89077
## 2 2001  -106.82933
## 3 2002    37.23536
## 4 2003    96.40155
## 5 2004    69.13959
## 6 2005    36.51212
# Para el PIB

pib <- WDI(
  country = 'MX',
  indicator = 'NY.GDP.MKTP.KD.ZG',
  start = 2000,
  end = 2025
)

pib <- data.frame(
  anio = pib$year,
  crecimiento_pib = pib$NY.GDP.MKTP.KD.ZG
)

head(pib)
##   anio crecimiento_pib
## 1 2025       0.5616835
## 2 2024       1.3505763
## 3 2023       3.1067964
## 4 2022       3.7097571
## 5 2021       6.0484835
## 6 2020      -8.3540346
datos <- merge(
  rend_anual,
  pib,
  by = "anio"
)

datos <- datos[order(datos$anio), ]

head(datos)
##   anio rendimiento crecimiento_pib
## 1 2000   -58.89077       5.0292840
## 2 2001  -106.82933      -0.4508453
## 3 2002    37.23536      -0.2365880
## 4 2003    96.40155       1.1855449
## 5 2004    69.13959       3.5654410
## 6 2005    36.51212       2.1132471
ggplot(datos, aes(x = anio)) +

  geom_col(
    aes(y = rendimiento, fill = "Rendimiento anual de GMEXICOB"),
    alpha = 1
  ) +

  geom_line(
    aes(y = crecimiento_pib, color = "Crecimiento del PIB"),
    linewidth = 1
  ) +

  geom_point(
    aes(y = crecimiento_pib, color = "Crecimiento del PIB"),
    size = 2
  ) +

  scale_fill_manual(
    values = c(
      "Rendimiento anual de GMEXICOB" = "light blue"
    )
  ) +

  scale_color_manual(
    values = c(
      "Crecimiento del PIB" = "dark blue"
    )
  ) +

  labs(
    title = "Rendimiento anual de GMEXICOB y crecimiento del PIB de México",
    x = "Año",
    y = "Porcentaje (%)",
    fill = "",
    color = ""
  ) +

  theme_classic()
